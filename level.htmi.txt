    <script>
        function calculateBMI() {
            const heightInput = document.getElementById('height').value;
            const weightInput = document.getElementById('weight').value;
            const resultContainer = document.getElementById('resultContainer');
            const bmiValueDisplay = document.getElementById('bmiValue');
            const bmiStatusDisplay = document.getElementById('bmiStatus');

            const height = parseFloat(heightInput);
            const weight = parseFloat(weightInput);

            if (!height || !weight || height <= 0 || weight <= 0) {
                alert('請輸入有效的身高與體重！');
                return;
            }

            // 身高公分轉公尺
            const heightInMeters = height / 100;
            // BMI 計算公式：體重 (kg) / (身高 (m) * 身高 (m))
            const bmi = (weight / (heightInMeters * heightInMeters)).toFixed(1);

            bmiValueDisplay.textContent = bmi;

            // 清除之前的狀態樣式
            bmiStatusDisplay.className = 'bmi-status';

            // 判斷 BMI 區間（依衛福部標準）
            if (bmi < 18.5) {
                bmiStatusDisplay.textContent = '體重過輕';
                bmiStatusDisplay.classList.add('underweight');
            } else if (bmi >= 18.5 && bmi < 24) {
                bmiStatusDisplay.textContent = '正常範圍';
                bmiStatusDisplay.classList.add('normal');
            } else if (bmi >= 24 && bmi < 27) {
                bmiStatusDisplay.textContent = '過重';
                bmiStatusDisplay.classList.add('overweight');
            } else {
                bmiStatusDisplay.textContent = '肥胖';
                bmiStatusDisplay.classList.add('obese');
            }

            resultContainer.style.display = 'block';
        }
    </script>
